namespace DungeonToolkit.Coyote;

public class WaveQueue  
{
    public bool IsEnabled { get; set; }
    public WavePlaybackMode PlaybackMode { get; set; }
    public WavePlayer? PlayingWave { get; }
    public PulseData CurrentPulseData { get; }
    public int NextWaveInterval { get; set; }
    public event EventHandler<PlayingWaveChangedEventArgs>? PlayingWaveChanged;
    public void Add(WavePlayer wavePlayer);
    public bool Remove(WavePlayer wavePlayer);
    public void Replace(IEnumerable<WavePlayer> wavePlayers);
    public void MoveUp(WavePlayer wavePlayer);
    public void Swap(WavePlayer a, WavePlayer b);
    public void Play(WavePlayer wavePlayer);
    public WavePlayer? NextWave();
    public PulseData NextPulseData();
    public class PlayingWaveChangedEventArgs(WavePlayer? wavePlayer) : EventArgs 
    {
        public WavePlayer? WavePlayer { get; }
    }
}
