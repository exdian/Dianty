namespace DungeonToolkit.Coyote;

public class WaveQueue  
{
    public bool IsEnabled { get; set; }
    public WavePlayMode PlayMode { get; set; }
    public Wave? CurrentWave { get; }
    public PulseData CurrentPulseData { get; }
    public int NextWaveInterval { get; set; }
    public event EventHandler<PlayingWaveChangedEventArgs>? PlayingWaveChanged;
    public void Add(Wave wave);
    public bool Remove(Wave wave);
    public void Replace(IEnumerable<WavePlayer> wavePlayers);
    public void MoveUp(Wave wave);
    public void Swap(Wave a, Wave b);
    public void Play(Wave wave);
    public void SetSpeed(Wave wave, int speed);
    public Wave? NextWave();
    public PulseData NextPulseData();
    public class PlayingWaveChangedEventArgs(Wave? wave) : EventArgs 
    {
        public Wave? Wave { get; }
    }
}
